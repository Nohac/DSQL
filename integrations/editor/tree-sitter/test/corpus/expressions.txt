query ExpressionMatrix(
  %value = 1
  %values = [1, 2, 3,]
  $field? = null
) {
  title(
    where not (.id %comparison[==, !=,] %value)
      or .id in %values
      and .name not in ["x", "y"]
      and .note is null
      and .kind is not null
      and exists .cast_info(filter visible when $:tenant where .note != null)
      and exists catalog::title
      and (.cast_info | count) >= 2
      and ~title.id == ..parent.id
      and %search on searchable
    order by %sort on selected_indexed
  ) {
    id
  }
}
