fragment Bits(%page? = null $structured? = null) on title {
  id
}

query SelectionMatrix(%page = 0 %direction = "desc" %options = {}) {
  ...Bits(%page <- %page, $structured, %, $) @.deprecated(reason: "binding")
  ...Bits @.deprecated(reason: "bare")
  ...movie: movies->title(
    filter visible when $:tenant
    where .production_year >= %year
      and .title like %search
    order by catalog::production_year %direction,
      %selected on selected,
      %indexed on indexed,
      %order on selected_indexed,
      %search on searchable,
    limit %page
    offset 1
  ) @dsql.deprecated(reason: "field") {
    ...aliases {
      id
    }
    cast_info->title
  }
  ratings | stats by year: .production_year, .kind_id {
    total: count,
    max .rating,
  }
}
